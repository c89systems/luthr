cmake_minimum_required(VERSION 3.25)
project(luthr_consumer LANGUAGES C)
find_package(luthr 0.1 REQUIRED CONFIG)
if(NOT luthr_TARGET STREQUAL LUTHR_TARGET OR NOT luthr_LUA_VERSION STREQUAL "5.5.1")
  message(FATAL_ERROR "Installed SDK target or Lua baseline does not match the selected consumer")
endif()
include(CheckCSourceCompiles)
set(CMAKE_TRY_COMPILE_TARGET_TYPE EXECUTABLE)
set(CMAKE_REQUIRED_LIBRARIES luthr::shared)
check_c_source_compiles("extern unsigned long luaE_initnativestack(void);
  int main(void) { return (int)luaE_initnativestack(); }" LUTHR_PRIVATE_SYMBOL_VISIBLE)
if(LUTHR_PRIVATE_SYMBOL_VISIBLE)
  message(FATAL_ERROR "Private Lua implementation symbol linked through the public shared SDK")
endif()
include("${LUTHR_SOURCE_DIR}/cmake/LocalRuntime.cmake")
enable_testing()
foreach(linkage IN ITEMS static shared)
  add_executable(consumer_${linkage} main.c)
  target_compile_options(consumer_${linkage} PRIVATE
    -std=c89 -Wall -Wextra -Wpedantic -pedantic-errors -Werror)
  target_link_libraries(consumer_${linkage} PRIVATE luthr::${linkage})
  luthr_local_runtime(consumer_${linkage})
  add_test(NAME consumer_${linkage} COMMAND consumer_${linkage})
  set_tests_properties(consumer_${linkage} PROPERTIES TIMEOUT 10)
endforeach()
add_library(native_fixture MODULE native_module.c)
set_target_properties(native_fixture PROPERTIES PREFIX "" C_STANDARD 99)
target_compile_options(native_fixture PRIVATE -Wall -Wextra -Werror)
target_link_libraries(native_fixture PRIVATE luthr::shared)
add_test(NAME shared_native_module COMMAND consumer_shared "$<TARGET_FILE_DIR:native_fixture>")
set_tests_properties(shared_native_module PROPERTIES TIMEOUT 10)

find_package(PkgConfig REQUIRED)
pkg_check_modules(LUTHR_PC REQUIRED luthr=0.1.0)
pkg_get_variable(pc_target luthr target)
if(NOT pc_target STREQUAL LUTHR_TARGET)
  message(FATAL_ERROR "pkg-config SDK target mismatch")
endif()
foreach(linkage IN ITEMS static shared)
  add_executable(pkg_consumer_${linkage} main.c)
  target_compile_options(pkg_consumer_${linkage} PRIVATE
    -std=c89 -Wall -Wextra -Wpedantic -pedantic-errors -Werror ${LUTHR_PC_CFLAGS_OTHER})
  target_include_directories(pkg_consumer_${linkage} PRIVATE ${LUTHR_PC_INCLUDE_DIRS})
  if(linkage STREQUAL "static")
    set(flags ${LUTHR_PC_STATIC_LDFLAGS})
    list(TRANSFORM flags REPLACE "^-lluthr$" "${LUTHR_PC_LIBDIR}/libluthr.a")
  else()
    set(flags ${LUTHR_PC_LDFLAGS})
  endif()
  target_link_libraries(pkg_consumer_${linkage} PRIVATE ${flags})
  set_target_properties(pkg_consumer_${linkage} PROPERTIES BUILD_RPATH "${LUTHR_PC_LIBDIR}")
  luthr_local_runtime(pkg_consumer_${linkage})
  add_test(NAME pkg_consumer_${linkage} COMMAND pkg_consumer_${linkage})
  set_tests_properties(pkg_consumer_${linkage} PROPERTIES TIMEOUT 10)
endforeach()
