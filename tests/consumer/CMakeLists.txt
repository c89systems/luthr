cmake_minimum_required(VERSION 3.25)
project(luthrum_consumer LANGUAGES C)
find_package(luthrum 0.1 REQUIRED CONFIG)
if(NOT luthrum_TARGET STREQUAL LUTHRUM_TARGET OR NOT luthrum_LUA_VERSION STREQUAL "5.5.1")
  message(FATAL_ERROR "Installed SDK target or Lua baseline does not match the selected consumer")
endif()
include(CheckCSourceCompiles)
set(CMAKE_TRY_COMPILE_TARGET_TYPE EXECUTABLE)
set(CMAKE_REQUIRED_LIBRARIES luthrum::shared)
check_c_source_compiles("extern unsigned long luaE_initnativestack(void);
  int main(void) { return (int)luaE_initnativestack(); }" LUTHRUM_PRIVATE_SYMBOL_VISIBLE)
if(LUTHRUM_PRIVATE_SYMBOL_VISIBLE)
  message(FATAL_ERROR "Private Lua implementation symbol linked through the public shared SDK")
endif()
include("${LUTHRUM_SOURCE_DIR}/cmake/LocalRuntime.cmake")
enable_testing()
foreach(linkage IN ITEMS static shared)
  add_executable(consumer_${linkage} main.c)
  target_compile_options(consumer_${linkage} PRIVATE
    -std=c89 -Wall -Wextra -Wpedantic -pedantic-errors -Werror)
  target_link_libraries(consumer_${linkage} PRIVATE luthrum::${linkage})
  luthrum_local_runtime(consumer_${linkage})
  add_test(NAME consumer_${linkage} COMMAND consumer_${linkage})
  set_tests_properties(consumer_${linkage} PROPERTIES TIMEOUT 10)
endforeach()
add_library(native_fixture MODULE native_module.c)
set_target_properties(native_fixture PROPERTIES PREFIX "" C_STANDARD 99)
target_compile_options(native_fixture PRIVATE -Wall -Wextra -Werror)
target_link_libraries(native_fixture PRIVATE luthrum::shared)
add_test(NAME shared_native_module COMMAND consumer_shared "$<TARGET_FILE_DIR:native_fixture>")
set_tests_properties(shared_native_module PROPERTIES TIMEOUT 10)

# Compile the released binding through the relocated SDK's native headers.
set(json_source "${LUTHRUM_SOURCE_DIR}/.cache/modules/lonejson-0.50.0")
set(json_sdk "${LUTHRUM_SOURCE_DIR}/.cache/modules/liblonejson-0.50.0-${LUTHRUM_TARGET}")
find_package(lonejson 0.50.0 EXACT CONFIG REQUIRED
  PATHS "${json_sdk}/lib/cmake/lonejson" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
get_target_property(lua_includes luthrum::shared INTERFACE_INCLUDE_DIRECTORIES)
add_library(lonejson_native MODULE "${json_source}/src/lua/lonejson_lua.c")
set_target_properties(lonejson_native PROPERTIES PREFIX "" OUTPUT_NAME core
  C_STANDARD 90 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules/lonejson")
target_include_directories(lonejson_native PRIVATE ${lua_includes})
target_compile_options(lonejson_native PRIVATE -Wall -Wextra -Werror)
# Lua symbols resolve from the embedding host's one runtime, for both linkages.
target_link_libraries(lonejson_native PRIVATE lonejson::lonejson_static m)
add_test(NAME lonejson_runtime_provider COMMAND "${CMAKE_COMMAND}"
  "-Dmodule=$<TARGET_FILE:lonejson_native>" "-Dnm=${CMAKE_NM}"
  "-Dreadelf=${CMAKE_READELF}" -P "${LUTHRUM_SOURCE_DIR}/tests/modules/ModuleRuntime.cmake")
set_tests_properties(lonejson_runtime_provider PROPERTIES TIMEOUT 10)
foreach(linkage IN ITEMS static shared)
  add_executable(lonejson_${linkage} lonejson.c)
  target_compile_options(lonejson_${linkage} PRIVATE
    -std=c89 -Wall -Wextra -Wpedantic -pedantic-errors -Werror)
  target_link_libraries(lonejson_${linkage} PRIVATE luthrum::${linkage})
  if(linkage STREQUAL "static")
    target_link_options(lonejson_${linkage} PRIVATE -Wl,--export-dynamic)
  endif()
  luthrum_local_runtime(lonejson_${linkage})
  add_test(NAME lonejson_installed_${linkage} COMMAND lonejson_${linkage}
    "${json_source}/lua" "${CMAKE_CURRENT_BINARY_DIR}/modules"
    "${LUTHRUM_SOURCE_DIR}/tests/modules/lonejson_semantics.lua")
  set_tests_properties(lonejson_installed_${linkage} PROPERTIES TIMEOUT 30)
endforeach()

find_package(PkgConfig REQUIRED)
pkg_check_modules(LUTHRUM_PC REQUIRED luthrum=0.1.0)
pkg_get_variable(pc_target luthrum target)
if(NOT pc_target STREQUAL LUTHRUM_TARGET)
  message(FATAL_ERROR "pkg-config SDK target mismatch")
endif()
foreach(linkage IN ITEMS static shared)
  add_executable(pkg_consumer_${linkage} main.c)
  target_compile_options(pkg_consumer_${linkage} PRIVATE
    -std=c89 -Wall -Wextra -Wpedantic -pedantic-errors -Werror ${LUTHRUM_PC_CFLAGS_OTHER})
  target_include_directories(pkg_consumer_${linkage} PRIVATE ${LUTHRUM_PC_INCLUDE_DIRS})
  if(linkage STREQUAL "static")
    set(flags ${LUTHRUM_PC_STATIC_LDFLAGS})
    list(TRANSFORM flags REPLACE "^-lluthrum$" "${LUTHRUM_PC_LIBDIR}/libluthrum.a")
  else()
    set(flags ${LUTHRUM_PC_LDFLAGS})
  endif()
  target_link_libraries(pkg_consumer_${linkage} PRIVATE ${flags})
  set_target_properties(pkg_consumer_${linkage} PROPERTIES BUILD_RPATH "${LUTHRUM_PC_LIBDIR}")
  luthrum_local_runtime(pkg_consumer_${linkage})
  add_test(NAME pkg_consumer_${linkage} COMMAND pkg_consumer_${linkage})
  set_tests_properties(pkg_consumer_${linkage} PROPERTIES TIMEOUT 10)
endforeach()
